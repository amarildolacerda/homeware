# TCP Radio Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement TCP radio nodes that communicate with the hub via HTTP REST over WiFi with UDP discovery, extending range for nodes in WiFi-only locations.

**Architecture:** Add `TcpRadioHandler` class implementing `RadioInterface` to the hub (UDP server + HTTP endpoints), and create a new `nodes/tcp/` node that discovers hub via UDP and sends sensor data via HTTP POST.

**Tech Stack:** ArduinoJson, ESP8266/ESP32 WiFi, UDP, HTTPClient, WebServer, shared library (RadioInterface, sensor_registry)

## Global Constraints

- `TCP_ENABLED` guards all TCP code in hub (compilation conditional)
- Loop non-blocking (no `delay()`, rule 15)
- WiFi non-blocking (rule 26)
- Dashboard standard (sidebar 180px, stats-header, polling 3s, rule 28)
- API endpoints mandatory: `/api/state`, `/api/settings`, `/api/wifi`, `/api/ota`, `/api/restart` (rule 29)
- Console/telnet mandatory via `common_console.h` (rule 30)
- EEPROM_SIZE = 512 (rule 31)
- Validate EEPROM strings (rule 20)
- `FW_VERSION` = current tag (rule 13)
- `lib_extra_dirs` pointing to `../../shared` (rule 14)
- UDP discovery reuses port 5000 from bridge
- Fallback to fixed IP if UDP discovery fails

```cpp
#ifndef TCP_PROTOCOL_H
#define TCP_PROTOCOL_H

#include <stdint.h>

// TCP Radio message types
#define TCP_MSG_SENSOR_DATA   0x01
#define TCP_MSG_PAIR_REQUEST  0x02
#define TCP_MSG_PAIR_RESPONSE 0x03
#define TCP_MSG_HEARTBEAT     0x05
#define TCP_MSG_COMMAND       0x07
#define TCP_MSG_TIME_SYNC     0x08
#define TCP_MSG_GW_ANNOUNCE   0x09
#define TCP_MSG_GW_DISCOVER   0x0A
#define TCP_MSG_RESTART       0x0C

// UDP Discovery port (same as bridge)
#define TCP_UDP_PORT 5000
#define TCP_HTTP_PORT 80

// Command queue TTL
#define TCP_COMMAND_TTL_MS 30000

// Max pending commands per device
#define TCP_MAX_PENDING_COMMANDS 10

// Max TCP clients (for reference, not used in HTTP mode)
#define TCP_MAX_CLIENTS 4

// UDP Discovery structures
struct __attribute__((packed)) tcp_gw_discover_t {
    uint8_t msg_type;     // TCP_MSG_GW_DISCOVER (0x0A)
    uint8_t sensor_type;  // sensor type
    char device_name[32]; // device name
};

struct __attribute__((packed)) tcp_gw_announce_t {
    uint8_t msg_type;      // TCP_MSG_GW_ANNOUNCE (0x09)
    uint8_t fw_version[4]; // firmware version
    char hub_ip[16];       // hub IP address
    uint16_t hub_port;     // HTTP port (80)
};

#endif
```

```cpp
#ifdef TCP_ENABLED

#include "tcp_radio_handler.h"
#include "sensor_registry.h"
#include "common_console.h"
#include "web_server.h"
#include <ESP8266WebServer.h>

extern ESP8266WebServer s_server;

TcpRadioHandler* TcpRadioHandler::s_self = nullptr;

TcpRadioHandler::TcpRadioHandler() {
    s_self = this;
}

int TcpRadioHandler::init() {
    // Initialize UDP server
    m_udp.begin(TCP_UDP_PORT);
    console.printf("[tcp] UDP server started on port %d\n", TCP_UDP_PORT);

    // Register HTTP endpoints
    s_server.on("/node/register", HTTP_POST, [this]() {
        if (!s_server.hasArg("plain")) {
            s_server.send(400, "application/json", "{\"error\":\"no body\"}");
            return;
        }
        
        DynamicJsonDocument doc(512);
        DeserializationError error = deserializeJson(doc, s_server.arg("plain"));
        if (error) {
            s_server.send(400, "application/json", "{\"error\":\"invalid json\"}");
            return;
        }
        
        const char* device_id = doc["device_id"];
        const char* device_name = doc["device_name"];
        const char* fw_version = doc["fw_version"];
        uint8_t sensor_type = doc["sensor_type"];
        
        if (!device_id || !device_name) {
            s_server.send(400, "application/json", "{\"error\":\"missing fields\"}");
            return;
        }
        
        // Validate device_id (printable chars 0x20-0x7E, rule 20)
        for (int i = 0; device_id[i]; i++) {
            if (device_id[i] < 0x20 || device_id[i] > 0x7E) {
                s_server.send(400, "application/json", "{\"error\":\"invalid device_id\"}");
                return;
            }
        }
        
        // Validate sensor_type (1-10)
        if (sensor_type < 1 || sensor_type > 10) {
            s_server.send(400, "application/json", "{\"error\":\"invalid sensor_type\"}");
            return;
        }
        
        // Get MAC from WiFi client
        uint8_t mac[6];
        WiFi.macAddress(mac);
        
        handle_register(mac, device_id, sensor_type, device_name, fw_version);
        
        // Find assigned slot
        int slot = find_slot_by_device_id(device_id);
        
        DynamicJsonDocument response(256);
        response["status"] = "ok";
        response["assigned_slot"] = slot;
        response["device_id"] = device_id;
        
        String responseStr;
        serializeJson(response, responseStr);
        s_server.send(200, "application/json", responseStr);
    });
    
    s_server.on("/node/state", HTTP_POST, [this]() {
        if (!s_server.hasArg("plain")) {
            s_server.send(400, "application/json", "{\"error\":\"no body\"}");
            return;
        }
        
        DynamicJsonDocument doc(512);
        DeserializationError error = deserializeJson(doc, s_server.arg("plain"));
        if (error) {
            s_server.send(400, "application/json", "{\"error\":\"invalid json\"}");
            return;
        }
        
        const char* device_id = doc["device_id"];
        if (!device_id) {
            s_server.send(400, "application/json", "{\"error\":\"missing device_id\"}");
            return;
        }
        
        JsonObject state = doc.as<JsonObject>();
        handle_state(device_id, state);
        
        s_server.send(200, "application/json", "{\"status\":\"ok\"}");
    });
    
    s_server.on("/node/heartbeat", HTTP_POST, [this]() {
        if (!s_server.hasArg("plain")) {
            s_server.send(400, "application/json", "{\"error\":\"no body\"}");
            return;
        }
        
        DynamicJsonDocument doc(256);
        DeserializationError error = deserializeJson(doc, s_server.arg("plain"));
        if (error) {
            s_server.send(400, "application/json", "{\"error\":\"invalid json\"}");
            return;
        }
        
        const char* device_id = doc["device_id"];
        if (!device_id) {
            s_server.send(400, "application/json", "{\"error\":\"missing device_id\"}");
            return;
        }
        
        handle_heartbeat(device_id);
        
        s_server.send(200, "application/json", "{\"status\":\"ok\"}");
    });
    
    s_server.on("/node/command/", HTTP_GET, [this]() {
        // Extract device_id from URL: /node/command/{device_id}
        String uri = s_server.uri();
        String device_id = uri.substring(uri.lastIndexOf("/") + 1);
        
        if (device_id.length() == 0) {
            s_server.send(400, "application/json", "{\"error\":\"missing device_id\"}");
            return;
        }
        
        DynamicJsonDocument response(256);
        handle_command_get(device_id.c_str(), response);
        
        String responseStr;
        serializeJson(response, responseStr);
        s_server.send(200, "application/json", responseStr);
    });
    
    console.println("[tcp] HTTP endpoints registered");
    return 0;
}

bool TcpRadioHandler::send_command(const uint8_t* mac, uint8_t state) {
    // Find device by MAC
    int slot = sensor_registry_find_by_mac(mac);
    if (slot < 0) return false;
    
    virtual_sensor_t* sensor = sensor_registry_get(slot);
    if (!sensor || !sensor->paired) return false;
    
    // Queue command
    PendingCommand cmd;
    cmd.command = (state == 1) ? "on" : "off";
    cmd.slot = slot;
    cmd.created_at = millis();
    
    m_pending_commands[sensor->bridge_device_id].push_back(cmd);
    
    console.printf("[tcp] Command queued for %s: %s\n", sensor->bridge_device_id, cmd.command.c_str());
    return true;
}

bool TcpRadioHandler::send_restart(const uint8_t* mac) {

    virtual_sensor_t* sensor = sensor_registry_get(slot);
    if (!sensor || !sensor->paired) return false;

    return true;
}

void TcpRadioHandler::handle_register(const uint8_t* mac, const char* device_id, uint8_t sensor_type, const char* device_name, const char* fw_version) {
    // Check if already registered
    int slot = find_slot_by_device_id(device_id);
    if (slot >= 0) {
        console.printf("[tcp] Device %s already registered at slot %d\n", device_id, slot);
        return;
    }
    
    // Find free slot
    slot = sensor_registry_find_free_slot();
    if (slot < 0) {
        console.println("[tcp] No free slots available");
        return;
    }
    
    // Add to registry
    sensor_registry_add(mac, sensor_type, slot, device_name, HW_CHIP_UNKNOWN, RADIO_TCP);
    
    virtual_sensor_t* sensor = sensor_registry_get(slot);
    if (sensor) {
        strncpy(sensor->bridge_device_id, device_id, sizeof(sensor->bridge_device_id) - 1);
        sensor->paired = true;
        sensor->online = true;
        sensor->last_seen = millis();
    }
    
    console.printf("[tcp] Device %s registered at slot %d (type %d)\n", device_id, slot, sensor_type);
}

void TcpRadioHandler::handle_state(const char* device_id, JsonObject& state) {
    int slot = find_slot_by_device_id(device_id);
    if (slot < 0) {
        console.printf("[tcp] Unknown device: %s\n", device_id);
        return;
    }
    
    virtual_sensor_t* sensor = sensor_registry_get(slot);
    if (!sensor) return;
    
    // Update state based on sensor type
    switch (sensor->type) {
        case SENSOR_TYPE_TEMP_HUM:
            if (state.containsKey("temperature")) sensor->state.temp_hum.temperature = state["temperature"];
            if (state.containsKey("humidity")) sensor->state.temp_hum.humidity = state["humidity"];
            break;
        case SENSOR_TYPE_GAS:
            if (state.containsKey("gas_level")) sensor->state.gas.gas_level = state["gas_level"];
            if (state.containsKey("alarm")) sensor->state.gas.alarm = state["alarm"];
            break;
        case SENSOR_TYPE_ONOFF:
            if (state.containsKey("state")) sensor->state.onoff.state = state["state"];
            break;
        default:
            break;
    }
    
    sensor->last_seen = millis();
    sensor->online = true;
    m_rx_count++;
    
    // TODO: Call queue_bridge_state(slot) if available
}

void TcpRadioHandler::handle_heartbeat(const char* device_id) {
    int slot = find_slot_by_device_id(device_id);
    if (slot < 0) return;
    
    virtual_sensor_t* sensor = sensor_registry_get(slot);
    if (!sensor) return;
    
    sensor->last_seen = millis();
    sensor->online = true;
    m_rx_count++;
}

bool TcpRadioHandler::handle_command_get(const char* device_id, JsonObject& response) {
    auto it = m_pending_commands.find(device_id);
    if (it == m_pending_commands.end() || it->second.empty()) {
        return false;
    }
    
    PendingCommand cmd = it->second.front();
    it->second.erase(it->second.begin());
    
    response["command"] = cmd.command;
    response["slot"] = cmd.slot;
    
    return true;
}

void TcpRadioHandler::handle_udp_discover() {
    int packetSize = m_udp.parsePacket();
    if (packetSize <= 0) return;
    
    uint8_t buf[64];
    int len = m_udp.read(buf, sizeof(buf));
    if (len < 1) return;
    
    uint8_t msg_type = buf[0];
    if (msg_type != TCP_MSG_GW_DISCOVER) return;
    
    // Parse discover message
    tcp_gw_discover_t* discover = (tcp_gw_discover_t*)buf;
    
    console.printf("[tcp] UDP Discover from %s (type %d)\n", 
                   discover->device_name, discover->sensor_type);
    
    // Send announce response
    tcp_gw_announce_t announce;
    announce.msg_type = TCP_MSG_GW_ANNOUNCE;
    
    // Get firmware version
    // TODO: Get from shared_config
    announce.fw_version[0] = 0;
    announce.fw_version[1] = 0;
    announce.fw_version[2] = 1;
    announce.fw_version[3] = 0;
    
    // Get hub IP
    IPAddress ip = WiFi.localIP();
    snprintf(announce.hub_ip, sizeof(announce.hub_ip), "%d.%d.%d.%d", 
             ip[0], ip[1], ip[2], ip[3]);
    announce.hub_port = TCP_HTTP_PORT;
    
    // Send unicast response
    m_udp.beginPacket(m_udp.remoteIP(), m_udp.remotePort());
    m_udp.write((uint8_t*)&announce, sizeof(announce));
    m_udp.endPacket();
    
    console.printf("[tcp] Sent announce to %s\n", m_udp.remoteIP().toString().c_str());
}

void TcpRadioHandler::cleanup_expired_commands() {
    unsigned long now = millis();
    for (auto it = m_pending_commands.begin(); it != m_pending_commands.end(); ) {
        auto& cmds = it->second;
        cmds.erase(
            std::remove_if(cmds.begin(), cmds.end(), [now](const PendingCommand& cmd) {
                return (now - cmd.created_at) > TCP_COMMAND_TTL_MS;
            }),
            cmds.end()
        );
        
        if (cmds.empty()) {
            it = m_pending_commands.erase(it);
        } else {
            ++it;
        }
    }
}

int TcpRadioHandler::find_slot_by_device_id(const char* device_id) {
    for (int i = 0; i < MAX_VIRTUAL_SENSORS; i++) {
        virtual_sensor_t* sensor = sensor_registry_get(i);
        if (sensor && sensor->paired && strcmp(sensor->bridge_device_id, device_id) == 0) {
            return i;
        }
    }
    return -1;
}
```

```cpp
void send_udp_discover() {
    tcp_gw_discover_t discover;
    discover.msg_type = TCP_MSG_GW_DISCOVER;
    discover.sensor_type = SENSOR_TYPE;
    
    // Get device name
    String hostname = WiFi.hostname();
    strncpy(discover.device_name, hostname.c_str(), sizeof(discover.device_name) - 1);
    
    // Broadcast to port 5000
    IPAddress broadcast = IPAddress(255, 255, 255, 255);
    s_udp.beginPacket(broadcast, TCP_UDP_PORT);
    s_udp.write((uint8_t*)&discover, sizeof(discover));
    s_udp.endPacket();
    
    console.printf("[tcp] Sent UDP discover (attempt %d)\n", s_discovery_retries);
}

void handle_udp_announce() {
    int packetSize = s_udp.parsePacket();
    if (packetSize <= 0) return;
    
    uint8_t buf[64];
    int len = s_udp.read(buf, sizeof(buf));
    if (len < 1) return;
    
    uint8_t msg_type = buf[0];
    if (msg_type != TCP_MSG_GW_ANNOUNCE) return;
    
    tcp_gw_announce_t* announce = (tcp_gw_announce_t*)buf;
    
    // Extract hub IP
    s_hub_ip = String(announce->hub_ip);
    s_hub_port = announce->hub_port;
    s_hub_found = true;
    
    console.printf("[tcp] Hub found: %s:%d\n", s_hub_ip.c_str(), s_hub_port);
    console.printf("[tcp] Hub FW: %d.%d.%d.%d\n", 
                   announce->fw_version[0], announce->fw_version[1],
                   announce->fw_version[2], announce->fw_version[3]);
}

bool send_to_hub(const char* endpoint, const String& payload) {
    if (WiFi.status() != WL_CONNECTED) return false;
    
    WiFiClient client;
    HTTPClient http;
    
    String url = "http://" + s_hub_ip + ":" + String(s_hub_port) + endpoint;
    
    if (http.begin(client, url)) {
        http.addHeader("Content-Type", "application/json");
        int httpCode = http.POST(payload);
        
        if (httpCode > 0) {
            console.printf("[tcp] %s -> %d\n", endpoint, httpCode);
            http.end();
            return httpCode == 200;
        } else {
            console.printf("[tcp] %s failed: %s\n", endpoint, http.errorToString(httpCode).c_str());
        }
        http.end();
    }
    
    return false;
}

bool register_with_hub() {
    DynamicJsonDocument doc(256);
    doc["device_id"] = String("agri_") + String(ESP.getChipId(), HEX);

    doc["device_name"] = WiFi.hostname().c_str();
    doc["fw_version"] = FW_VERSION;
    
    String payload;
    serializeJson(doc, payload);
    
    if (send_to_hub("/node/register", payload)) {
        s_registered = true;
        s_retry_count = 0;
        console.println("[tcp] Registered with hub");
        return true;
    }
    
    s_retry_count++;
    return false;
}

    // Add sensor-specific data
    switch (SENSOR_TYPE) {
        case SENSOR_TYPE_TEMP_HUM:

            break;

            break;

        s_retry_count = 0;

    }

// EEPROM functions for hub IP persistence
void save_hub_ip_to_eeprom(const String& ip) {
    EEPROM.begin(EEPROM_SIZE);
   EEPROM.put(EEPROM_HUB_IP_OFFSET, ip.c_str());
    EEPROM.write(EEPROM_HUB_IP_VALID, 0x01);
    EEPROM.commit();
    console.printf("[tcp] Hub IP saved to EEPROM: %s\n", ip.c_str());
}

String load_hub_ip_from_eeprom() {
    EEPROM.begin(EEPROM_SIZE);
    if (EEPROM.read(EEPROM_HUB_IP_VALID) != 0x01) {
        return "";  // Not configured
    }
    char buf[16];
    EEPROM.get(EEPROM_HUB_IP_OFFSET, buf);
    return String(buf);
}

void loop() {
    // WiFi loop (non-blocking)
    mywifi_loop();
    handle_wifi();
    
    // OTA
    ArduinoOTA.handle();
    
    // Web server
    s_server.handleClient();
    
    // Console
    console.loop();
    
    // Check WiFi connection
    if (WiFi.status() != WL_CONNECTED) {
        digitalWrite(LED_PIN, (millis() / LED_BLINK_WIFI_MS) % 2 ? LED_ON : LED_OFF);
        return;
    }
    
    digitalWrite(LED_PIN, LED_OFF);
    
    // Hub discovery logic
    if (!s_hub_found) {
        // Strategy 1: Try configured IP (from EEPROM)
        if (s_hub_ip_configured) {
            if (millis() - s_last_reconnect > RECONNECT_INTERVAL) {
                if (register_with_hub()) {
                    s_hub_found = true;
                    console.printf("[tcp] Connected to configured hub: %s\n", s_hub_ip.c_str());
                } else {
                    s_http_fallback_retries++;
                    if (s_http_fallback_retries > HUB_FALLBACK_RETRIES) {
                        // Configured IP not responding → trigger UDP discovery
                        console.println("[tcp] Configured hub not responding, starting UDP discovery...");
                        s_hub_ip_configured = false;
                        s_http_fallback_retries = 0;
                        s_discovery_retries = 0;
                    }
                }
                s_last_reconnect = millis();
            }
            return;
        }
        
        // Strategy 2: UDP discovery (no IP configured or fallback)
        if (millis() - s_last_discovery > DISCOVERY_INTERVAL) {
            send_udp_discover();
            s_last_discovery = millis();
            s_discovery_retries++;
            
            if (s_discovery_retries > MAX_DISCOVERY_RETRIES) {
                // Strategy 3: Final fallback to default IP
                s_hub_ip = HUB_IP_DEFAULT;
                s_hub_found = true;
                console.printf("[tcp] Using default hub IP: %s\n", s_hub_ip.c_str());
            }
        }
        
        // Check for UDP response
        handle_udp_announce();
        if (s_hub_found && !s_hub_ip_configured) {
            // Save discovered IP to EEPROM for next boot
            save_hub_ip_to_eeprom(s_hub_ip);
        }
        return;
    }
    
    // Registration retry
    if (!s_registered) {
        if (millis() - s_last_reconnect > RECONNECT_INTERVAL) {
            if (!register_with_hub()) {
                s_http_fallback_retries++;
                if (s_http_fallback_retries > HUB_FALLBACK_RETRIES) {
                    // Hub went offline → trigger UDP discovery
                    console.println("[tcp] Hub offline, restarting UDP discovery...");
                    s_hub_found = false;
                    s_registered = false;
                    s_http_fallback_retries = 0;
                    s_discovery_retries = 0;
                }
            }
            s_last_reconnect = millis();
        }
        return;
    }
    
    // Send state periodically
    if (millis() - s_last_state_send > STATE_UPDATE_INTERVAL) {
        send_state();
    }
    
    // Send heartbeat periodically
    if (millis() - s_last_heartbeat > HEARTBEAT_INTERVAL) {
        send_heartbeat();
    }
    
    // Check for commands
    check_commands();
}
```
